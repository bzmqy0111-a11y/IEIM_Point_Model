cmake_minimum_required(VERSION 3.16)
project(IEIMPointModel VERSION 1.0.0 LANGUAGES CXX)
add_library(ieim_model INTERFACE)
target_include_directories(ieim_model INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ieim_model INTERFACE cxx_std_17)
add_executable(reconstruct src/main.cpp)
target_link_libraries(reconstruct PRIVATE ieim_model)
if(MSVC)
  target_compile_options(reconstruct PRIVATE /W4 /permissive-)
else()
  target_compile_options(reconstruct PRIVATE -Wall -Wextra -pedantic)
endif()
add_executable(example_point examples/call_from_project.cpp)
target_link_libraries(example_point PRIVATE ieim_model)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME point_regression COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_point_model.py
      --exe $<TARGET_FILE:reconstruct> --data-dir ${CMAKE_CURRENT_SOURCE_DIR}/data)
  endif()
endif()
